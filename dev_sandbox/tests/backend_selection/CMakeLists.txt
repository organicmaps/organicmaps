cmake_minimum_required(VERSION 3.22.1)
project(sandbox_backend_selection LANGUAGES C CXX)

get_filename_component(OMIM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
option(BUILD_DEV_SANDBOX "Build the developer graphics sandbox when its dependencies are available" ON)
if (BUILD_DEV_SANDBOX)
  include(${OMIM_ROOT}/dev_sandbox/linux_backend.cmake)
else()
  message(STATUS "dev_sandbox: disabled by BUILD_DEV_SANDBOX=OFF")
  set(DEV_SANDBOX_WAYLAND_AVAILABLE FALSE)
endif()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/selected.txt "${DEV_SANDBOX_WAYLAND_AVAILABLE}")
if (NOT DEV_SANDBOX_WAYLAND_AVAILABLE)
  return()
endif()

set(GLFW_BUILD_DOCS OFF)
set(GLFW_BUILD_EXAMPLES OFF)
set(GLFW_BUILD_TESTS OFF)
set(GLFW_INSTALL OFF)
add_subdirectory(${OMIM_ROOT}/3party/glfw glfw)
set_target_properties(glfw PROPERTIES UNITY_BUILD OFF)
add_subdirectory(${OMIM_ROOT}/3party/imgui imgui)
add_executable(check_backend check_backend.cpp)
target_link_libraries(check_backend PRIVATE glfw)
enable_testing()
add_test(NAME compiled_backend COMMAND check_backend)
